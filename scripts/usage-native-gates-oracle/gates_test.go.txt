var errGateDial = errors.New("gate oracle: dial reached")

type gateFile struct {
	Path     string `json:"path"`
	Kind     string `json:"kind,omitempty"` // file (default), symlink, dir
	Contents string `json:"contents,omitempty"`
	Hex      string `json:"contents_hex,omitempty"` // exact bytes when Contents cannot carry them
	PadTo    int    `json:"pad_to,omitempty"`
	Target   string `json:"target,omitempty"`
}

type gateCase struct {
	ID            string            `json:"id"`
	IDs           []string          `json:"ids"`
	Env           map[string]string `json:"env"`
	Home          string            `json:"home,omitempty"` // set (default), unset, empty
	Files         []gateFile        `json:"files,omitempty"`
	Keychain      string            `json:"keychain,omitempty"`
	KeychainMs    int64             `json:"keychain_expires_ms,omitempty"`
	Status        int               `json:"status"`
	Nondetermined bool              `json:"nondeterministic,omitempty"`
}

type gateRequest struct {
	Method     string            `json:"method"`
	URL        string            `json:"url"`
	Host       string            `json:"host"`
	RequestURI string            `json:"request_uri"`
	Headers    map[string]string `json:"headers"`
	RawHeaders map[string]string `json:"raw_headers"`
	Body       string            `json:"body"`
}

type gateObserved struct {
	Report        json.RawMessage `json:"report"`
	Requests      []gateRequest   `json:"requests"`
	KeychainCalls int             `json:"keychain_calls"`
}

type gateRecorder struct {
	mu       sync.Mutex
	status   int
	requests []gateRequest
	inner    *http.Transport
}

// Go draws a random server-instance UUID per OpenCode request; hostname and runtime version are host facts.
var gateDynamicHeaders = map[string]bool{"x-msh-device-name": true, "x-msh-os-version": true, "x-msh-device-model": true, "x-server-instance": true}

func (r *gateRecorder) RoundTrip(req *http.Request) (*http.Response, error) {
	if _, err := r.inner.RoundTrip(req); !errors.Is(err, errGateDial) {
		if err == nil {
			return nil, errors.New("gate oracle: unexpected transport success")
		}
		return nil, err
	}
	headers := map[string]string{}
	rawHeaders := map[string]string{}
	for name, values := range req.Header {
		key := strings.ToLower(name)
		if key == "content-length" {
			continue
		}
		value := strings.Join(values, ", ")
		if !utf8.ValidString(value) {
			rawHeaders[key] = hex.EncodeToString([]byte(value))
			continue
		}
		if gateDynamicHeaders[key] {
			value = "<dynamic>"
		}
		headers[key] = value
	}
	body := ""
	if req.Body != nil {
		data, _ := io.ReadAll(req.Body)
		body = string(data)
	}
	r.mu.Lock()
	defer r.mu.Unlock()
	r.requests = append(r.requests, gateRequest{Method: req.Method, URL: req.URL.String(), Host: req.URL.Host, RequestURI: req.URL.RequestURI(), Headers: headers, RawHeaders: rawHeaders, Body: body})
	return &http.Response{StatusCode: r.status, Header: make(http.Header), Body: io.NopCloser(strings.NewReader("{}")), Request: req}, nil
}

func gateClearEnv(t *testing.T) {
	for _, pair := range os.Environ() {
		name, _, _ := strings.Cut(pair, "=")
		t.Setenv(name, "")
		if err := os.Unsetenv(name); err != nil {
			t.Fatal(err)
		}
	}
}

func gateApplyFiles(t *testing.T, home string, files []gateFile) {
	for _, f := range files {
		path := filepath.Join(home, f.Path)
		switch f.Kind {
		case "dir":
			if err := os.MkdirAll(path, 0o700); err != nil {
				t.Fatal(err)
			}
		case "symlink":
			if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
				t.Fatal(err)
			}
			if err := os.Symlink(strings.ReplaceAll(f.Target, "$HOME", home), path); err != nil {
				t.Fatal(err)
			}
		default:
			if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
				t.Fatal(err)
			}
			contents := f.Contents
			if f.Hex != "" {
				raw, err := hex.DecodeString(f.Hex)
				if err != nil {
					t.Fatal(err)
				}
				contents = string(raw)
			}
			if f.PadTo > len(contents) {
				contents += strings.Repeat(" ", f.PadTo-len(contents))
			}
			if err := os.WriteFile(path, []byte(contents), 0o600); err != nil {
				t.Fatal(err)
			}
		}
	}
}

func gateRun(t *testing.T, c gateCase, root string) gateObserved {
	gateClearEnv(t)
	if err := os.RemoveAll(root); err != nil {
		t.Fatal(err)
	}
	home := filepath.Join(root, "home")
	if err := os.MkdirAll(home, 0o700); err != nil {
		t.Fatal(err)
	}
	t.Chdir(home)
	for name, value := range c.Env {
		t.Setenv(name, value)
	}
	switch c.Home {
	case "unset":
	case "empty":
		t.Setenv("HOME", "")
	default:
		t.Setenv("HOME", home)
	}
	gateApplyFiles(t, home, c.Files)
	status := c.Status
	if status == 0 {
		status = 401
	}
	recorder := &gateRecorder{status: status, inner: &http.Transport{DialContext: func(context.Context, string, string) (net.Conn, error) { return nil, errGateDial }}}
	client := &http.Client{Transport: recorder, CheckRedirect: func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }}
	calls := 0
	keychain := func() (string, *time.Time) {
		calls++
		if c.Keychain == "" {
			return "", nil
		}
		var expires *time.Time
		if c.KeychainMs > 0 {
			value := time.UnixMilli(c.KeychainMs)
			expires = &value
		}
		return c.Keychain, expires
	}
	providers := providersForIDs(client, c.IDs, keychain, antigravityOracleProbe{})
	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
	defer cancel()
	report := BuildReport(ctx, providers)
	data, err := json.Marshal(report)
	if err != nil {
		t.Fatal(err)
	}
	return gateObserved{Report: data, Requests: recorder.requests, KeychainCalls: calls}
}

func itoa(i int) string { return strconv.Itoa(i) }

func b64url(s string) string {
	const alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_"
	data := []byte(s)
	var out []byte
	for i := 0; i < len(data); i += 3 {
		var chunk [3]byte
		n := copy(chunk[:], data[i:])
		v := uint(chunk[0])<<16 | uint(chunk[1])<<8 | uint(chunk[2])
		for j := 0; j < n+1; j++ {
			out = append(out, alphabet[(v>>(18-6*uint(j)))&63])
		}
	}
	return string(out)
}

func TestNativeGatesOracle768(t *testing.T) {
	output := os.Getenv("USAGE_768_GATES_OUT")
	scratch := os.Getenv("USAGE_768_GATES_ROOT")
	if output == "" || scratch == "" {
		t.Skip("USAGE_768_GATES_OUT and USAGE_768_GATES_ROOT are required")
	}
	type record struct {
		gateCase
		Observed []gateObserved `json:"observed"`
	}
	var records []record
	for index, c := range gateCases() {
		runs := 1
		if c.Nondetermined {
			runs = 40
		}
		seen := map[string]bool{}
		var observed []gateObserved
		for run := 0; run < runs; run++ {
			// Placeholders are resolved per run so $HOME in env values points at the run's own HOME.
			working := c
			working.Env = map[string]string{}
			root := filepath.Join(scratch, itoa(index)+"-"+itoa(run))
			for k, v := range c.Env {
				working.Env[k] = strings.ReplaceAll(v, "$HOME", filepath.Join(root, "home"))
			}
			result := gateRun(t, working, root)
			// Normalise the per-run root out of the encoded report.
			key := strings.ReplaceAll(string(result.Report), root, "<root>") + "|" + gateKey(result.Requests)
			if !seen[key] {
				seen[key] = true
				observed = append(observed, result)
			}
		}
		sort.SliceStable(observed, func(i, j int) bool { return string(observed[i].Report) < string(observed[j].Report) })
		records = append(records, record{gateCase: c, Observed: observed})
	}
	data, err := json.Marshal(map[string]any{"schema_version": 1, "go_version": runtime.Version(), "goos": runtime.GOOS, "goarch": runtime.GOARCH, "cases": records})
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(output, append(data, '\n'), 0o600); err != nil {
		t.Fatal(err)
	}
}

func gateKey(requests []gateRequest) string {
	data, _ := json.Marshal(requests)
	return string(data)
}
