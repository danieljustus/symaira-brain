const credential620 = "oracle-only-invalid-620-credential"

type request620 struct {
	Method  string            `json:"method"`
	URL     string            `json:"url"`
	Headers map[string]string `json:"headers"`
	Body    string            `json:"body"`
}
type reply620 struct {
	Status  int               `json:"status"`
	Body    string            `json:"body"`
	Headers map[string]string `json:"headers"`
}
type case620 struct {
	ID       string       `json:"id"`
	Variant  string       `json:"variant"`
	Reply    reply620     `json:"reply"`
	Report   Report       `json:"report"`
	Requests []request620 `json:"requests"`
	JSON     string       `json:"json,omitempty"`
	Table    string       `json:"table,omitempty"`
}
type transport620 struct {
	reply    reply620
	requests []request620
}

func (r *transport620) RoundTrip(req *http.Request) (*http.Response, error) {
	headers := map[string]string{}
	for name := range req.Header {
		value := req.Header.Get(name)
		if strings.EqualFold(name, "X-Server-Instance") {
			if !strings.HasPrefix(value, "server-fn:") {
				panic("unexpected OpenCode instance")
			}
			value = "server-fn:<random>"
		}
		headers[strings.ToLower(name)] = value
	}
	body := ""
	if req.Body != nil {
		data, err := io.ReadAll(req.Body)
		if err != nil {
			return nil, err
		}
		body = string(data)
	}
	r.requests = append(r.requests, request620{req.Method, req.URL.String(), headers, body})
	reply := r.reply
	if strings.HasSuffix(req.URL.Path, "/GetUnleashData") {
		reply = reply620{200, "{}", map[string]string{}}
	}
	responseHeaders := http.Header{}
	for name, value := range reply.Headers {
		responseHeaders.Set(name, value)
	}
	return &http.Response{StatusCode: reply.Status, Header: responseHeaders, Body: io.NopCloser(strings.NewReader(reply.Body)), Request: req}, nil
}

func provider620(variant string, client *http.Client) Provider {
	switch variant {
	case "antigravity":
		return newAntigravityProvider(antigravityOracleProbe{processes: "123 /Applications/Antigravity/language_server --app_data_dir antigravity --csrf_token fixture\n", portsByPID: map[int]string{123: "COMMAND PID NAME\nlanguage 123 TCP 127.0.0.1:43123 (LISTEN)\n"}}, client)
	case "claude-api":
		return &ClaudeProvider{adminKey: credential620, adminSource: "env", client: client}
	case "claude-oauth":
		return &ClaudeProvider{oauthToken: credential620, oauthSource: "env", client: client}
	case "claude-chain":
		return &ClaudeProvider{adminKey: credential620, adminSource: "env", oauthToken: credential620, oauthSource: "env", client: client}
	case "codex":
		return &CodexProvider{accessToken: credential620, credSource: "env", client: client}
	case "copilot", "copilot-enterprise", "copilot-enterprise-port":
		host := ""
		if variant == "copilot-enterprise" {
			host = "copilot.example.com"
		}
		if variant == "copilot-enterprise-port" {
			host = "copilot.example.com:8443"
		}
		return &CopilotProvider{accessToken: credential620, credSource: "env", enterpriseHost: host, client: client}
	case "cursor":
		return &CursorProvider{cookieHeader: credential620, credSource: "env", client: client}
	case "kimi-api":

	case "kimi-cli", "kimi-cli-no-device":
		device := "oracle-620-device"
		if variant == "kimi-cli-no-device" {
			device = ""
		}
		return &KimiProvider{cliAccessToken: credential620, cliDeviceID: device, baseURL: kimiDefaultAPIBase, client: client}
	case "kimi-web":
		return &KimiProvider{authToken: credential620, authSource: "env", baseURL: kimiDefaultAPIBase, client: client}

	panic("unknown fixture variant " + variant)

func TestUsageFetchOracle620(t *testing.T) {
	// Identity comes from the OS, even if a caller overrides this Windows env.
	t.Setenv("COMPUTERNAME", "intentionally-wrong-oracle-620-host")
	files := map[string]string{
		"antigravity": "antigravity-quota-summary.json",
		"claude-api":  "claude-admin-cost.json", "claude-oauth": "claude-oauth-usage.json", "claude-chain": "claude-admin-cost.json",
		"codex": "codex-wham-usage.json", "copilot": "copilot-user.json", "copilot-enterprise": "copilot-user.json", "copilot-enterprise-port": "copilot-user.json",
		"cursor": "cursor-usage-summary.json", "kimi-api": "kimi-fallback-api-usages.json", "kimi-cli": "kimi-fallback-api-usages.json", "kimi-cli-no-device": "kimi-fallback-api-usages.json", "kimi-web": "kimi-fallback-web-usages.json", "kimi-chain": "kimi-fallback-api-usages.json",
		"moonshot-ai": "moonshot-balance-ai.json", "moonshot-cn": "moonshot-balance-cn.json", "nous": "nous-account.json", "opencode": "opencode-subscription-no-reset.json", "openrouter": "openrouter-credits.json",
	}
	variants := []string{"claude-api", "claude-oauth", "claude-chain", "codex", "copilot", "copilot-enterprise", "copilot-enterprise-port", "cursor", "kimi-api", "kimi-cli", "kimi-cli-no-device", "kimi-web", "kimi-chain", "moonshot-ai", "moonshot-cn", "nous", "opencode", "openrouter", "antigravity"}
	out := []case620{}
	for _, variant := range variants {
		success, err := os.ReadFile(filepath.Join("testdata", files[variant]))
		if err != nil {
			t.Fatal(err)
		}
		responses := []struct {
			label string
			reply reply620
		}{
			{"success", reply620{200, string(success), map[string]string{}}},
			{"unauthorized", reply620{401, `{"error":"denied"}`, map[string]string{}}},
			{"forbidden", reply620{403, `{"error":"denied"}`, map[string]string{}}},
			{"rate-limit", reply620{429, `{"error":"slow"}`, map[string]string{"Retry-After": "17"}}},
			{"rate-limit-no-delay", reply620{429, `{"error":"slow"}`, map[string]string{}}},
			{"server-error", reply620{500, "oops", map[string]string{}}},
			{"malformed", reply620{200, "not json at all", map[string]string{}}},
		}
		for _, response := range responses {
			recorder := &transport620{reply: response.reply, requests: []request620{}}
			provider := provider620(variant, &http.Client{Transport: recorder})
			report := BuildReport(context.Background(), []Provider{provider})
			if response.label == "success" && report.Providers[0].Snapshot == nil {
				t.Fatalf("%s: invalid success fixture: %s", variant, report.Providers[0].Error)
			}
			if snapshot := report.Providers[0].Snapshot; snapshot != nil {
				snapshot.FetchedAt = time.Unix(0, 0).UTC()
			}
			out = append(out, case620{ID: variant + "/" + response.label, Variant: variant, Reply: response.reply, Report: report, Requests: recorder.requests})
		}
	}
	data, err := json.MarshalIndent(out, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("USAGE_FETCH_ORACLE_620"), append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
}
