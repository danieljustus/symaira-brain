type credentialTransport768 struct{ Headers []map[string]string }

func (r *credentialTransport768) RoundTrip(req *http.Request) (*http.Response, error) {
	headers := map[string]string{}
	for _, name := range []string{"Authorization", "X-Api-Key", "Cookie"} {
		if value := req.Header.Get(name); value != "" {
			headers[strings.ToLower(name)] = value
		}
	}
	r.Headers = append(r.Headers, headers)
	return &http.Response{StatusCode: 401, Body: io.NopCloser(strings.NewReader("{}")), Header: make(http.Header)}, nil
}
