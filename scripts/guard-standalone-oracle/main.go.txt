// Supplemental process entrypoint over unchanged public Go Guard commands.
// This file is not part of the production Go module or a frozen fixture.
package main

import (
    "fmt"
    "io"
    "os"
    "github.com/danieljustus/symaira-brain/guard/cmd/symguard/decide"
    "github.com/danieljustus/symaira-brain/guard/cmd/symguard/doctor"
    "github.com/danieljustus/symaira-brain/guard/cmd/symguard/grants"
    "github.com/danieljustus/symaira-brain/guard/cmd/symguard/scan"
    "github.com/danieljustus/symaira-brain/guard/cmd/symguard/version"
)

const usage = `symguard — local-first security gateway for AI agents

Usage:
  symguard <command> [flags]

Commands:
  version   Print version and build info
  doctor    Check system health and configuration
  decide    Read a JSON decision request from stdin, write the decision to stdout
  grants    List and revoke standing grants
  scan      Discover MCP servers across supported AI clients
  help      Show this help message

Run 'symguard <command> --help' for details on a specific command.`

func run(args []string, out, errOut io.Writer) int {
    if len(args) == 0 { fmt.Fprintln(out, usage); return 2 }
    switch args[0] {
    case "help", "--help", "-h": fmt.Fprintln(out, usage); return 0
    case "version": if version.Run(args[1:], out) != nil { return 1 }; return 0
    case "doctor": return doctor.Run(out)
    case "decide": return decide.Run(args[1:], os.Stdin, out, nil)
    case "scan": return scan.Run(args[1:], out, errOut)
    case "grants": grants.Run(args[1:], out); return 0
    default: fmt.Fprintf(errOut, "symguard: unknown command %q\n\n", args[0]); fmt.Fprintln(errOut, usage); return 2
    }
}
func main() { os.Exit(run(os.Args[1:], os.Stdout, os.Stderr)) }
