// Go-only actual constructor/source baseline; no native acceptance claim.
package usage

import (
	"bytes"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"runtime"
	"strings"
	"testing"
)

type localCase768 struct {
	ID       string            `json:"id"`
	Provider string            `json:"provider"`
	Files    map[string]string `json:"files"`
	Env      map[string]string `json:"env"`
	HomeMode string            `json:"home_mode"`
}
type localTransport768 struct{ Requests []any }

func (rt *localTransport768) RoundTrip(req *http.Request) (*http.Response, error) {
	body := []byte{}
	if req.Body != nil {
		body, _ = io.ReadAll(req.Body)
	}
	headers := map[string][]string{}
	headerHex := map[string][]string{}
	for name, values := range req.Header {
		headers[name] = values
		for _, value := range values {
			headerHex[name] = append(headerHex[name], hex.EncodeToString([]byte(value)))
		}
	}
	rt.Requests = append(rt.Requests, map[string]any{"method": req.Method, "url": req.URL.String(), "headers": headers, "header_value_hex": headerHex, "body_hex": hex.EncodeToString(body)})
	return &http.Response{StatusCode: 401, Body: io.NopCloser(strings.NewReader("{}")), Header: make(http.Header)}, nil
}
func TestUsageLocalFilesBaseline768(t *testing.T) {
	data, err := os.ReadFile(os.Getenv("USAGE_LOCAL_INPUT"))
	if err != nil {
		t.Fatal(err)
	}
	var cases []localCase768
	if err = json.Unmarshal(data, &cases); err != nil {
		t.Fatal(err)
	}
	if len(cases) != 97 {
		t.Fatal("complete 97-case Go baseline required")
	}
	records := []any{}
	for _, entry := range cases {
		root := filepath.Join(os.Getenv("USAGE_LOCAL_ROOT"), entry.ID)
		home := filepath.Join(root, "home")
		profile := home
		if entry.HomeMode == "different" {
			profile = filepath.Join(root, "profile")
		}
		for _, dir := range []string{home, profile} {
			if err = os.MkdirAll(dir, 0755); err != nil {
				t.Fatal(err)
			}
		}
		files := map[string][]byte{}
		for name, value := range entry.Files {
			contents, err := hex.DecodeString(value)
			if err != nil {
				t.Fatal(err)
			}
			path := filepath.Join(home, filepath.FromSlash(name))
			if err = os.MkdirAll(filepath.Dir(path), 0755); err != nil {
				t.Fatal(err)
			}
			if err = os.WriteFile(path, contents, 0600); err != nil {
				t.Fatal(err)
			}
			files[path] = contents
		}
		if entry.HomeMode == "different" {
			path := filepath.Join(profile, ".kimi-code", "credentials", "kimi-code.json")
			contents := []byte(`{"access_token":"profile-token"}`)
			if err = os.MkdirAll(filepath.Dir(path), 0755); err != nil {
				t.Fatal(err)
			}
			if err = os.WriteFile(path, contents, 0600); err != nil {
				t.Fatal(err)
			}
			files[path] = contents
		}
		for _, name := range []string{"ANTHROPIC_ADMIN_KEY", "ANTHROPIC_OAUTH_TOKEN", "CODEX_ACCESS_TOKEN", "COPILOT_ACCESS_TOKEN", "CURSOR_COOKIE", "KIMI_CODE_API_KEY", "KIMI_AUTH_TOKEN", "MOONSHOT_API_KEY", "NOUS_PORTAL_ACCESS_TOKEN", "OPENCODE_COOKIE", "OPENROUTER_API_KEY", "KIMI_CODE_HOME", "HERMES_HOME", "CODEX_HOME", "KIMI_CODE_BASE_URL", "HERMES_PORTAL_BASE_URL", "OPENROUTER_API_URL", "OPENCODE_WORKSPACE_ID", "USAGE_LOCAL_FILES_ABSENT"} {
			t.Setenv(name, "")
		}
		t.Setenv("HOME", home)
		t.Setenv("USERPROFILE", profile)
		for name, value := range entry.Env {
			t.Setenv(name, strings.ReplaceAll(value, "$HOME", home))
		}
		transport := &localTransport768{Requests: []any{}}
		client := &http.Client{Transport: transport}
		var provider Provider
		switch entry.Provider {
		case "copilot":
			provider = NewCopilotProvider(client)
		case "kimi":
			provider = NewKimiProvider(client)
		case "nous":
			provider = NewNousPortalProvider(client)
		case "opencode":
			provider = NewOpenCodeProvider(client)
		case "openrouter":
			provider = NewOpenRouterProvider(client)
		default:
			t.Fatal("unknown provider", entry.Provider)
		}
		report := BuildReport(context.Background(), []Provider{provider})
		hashes := map[string]string{}
		for path, before := range files {
			after, err := os.ReadFile(path)
			if err != nil || !bytes.Equal(before, after) {
				t.Fatal("Go mutated owned source", path, err)
			}
			sum := sha256.Sum256(before)
			hashes[path] = hex.EncodeToString(sum[:])
		}
		selected := map[string]any{}
		if entry.Provider == "kimi" {
			p := provider.(*KimiProvider)
			selected = map[string]any{"cli_home": kimiCLIHome(), "cli_token_hex": hex.EncodeToString([]byte(p.cliAccessToken)), "device_id_hex": hex.EncodeToString([]byte(p.cliDeviceID)), "base_url": p.baseURL}
		}
		if entry.Provider == "copilot" {
			p := provider.(*CopilotProvider)
			selected = map[string]any{"token_hex": hex.EncodeToString([]byte(p.accessToken)), "source": p.credSource}
		}
		records = append(records, map[string]any{"id": entry.ID, "provider": entry.Provider, "home": home, "userprofile": profile, "env": entry.Env, "file_sha256": hashes, "selected": selected, "report": report, "requests": transport.Requests})
	}
	output := map[string]any{"schema_version": 1, "status": "immutable-Go-only baseline; native successor unimplemented and unverified", "oracle_commit": "dcddcef0df5789123c7c9a7ebe6e01f10e941f2c", "go_runtime": runtime.Version(), "go_os": runtime.GOOS, "go_arch": runtime.GOARCH, "cases": 97, "read_only_cases": 97, "records": records}
	data, err = json.MarshalIndent(output, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(os.Getenv("USAGE_LOCAL_OUTPUT"), append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
}
